const bcrypt = require('bcrypt'); // Adicione no topo do arquivo

// Rota de CADASTRO
app.post('/api/cadastro', async (req, res) => {
    const { nome, email, senha } = req.body;

    try {
        // Criptografa a senha (o '10' é o custo do processamento do hash)
        const senhaHash = await bcrypt.hash(senha, 10);

        // Insere no banco. Assumindo que perfil_id 1 seja o 'Aluno' padrão
        const sql = 'INSERT INTO usuarios (nome, email, senha, perfil_id) VALUES (?, ?, ?, 1)';

        pool.query(sql, [nome, email, senhaHash], (err, result) => {
            if (err) {
                return res.status(500).json({ erro: 'Erro ao criar usuário ou email já existe.' });
            }
            res.json({ mensagem: 'Conta criada com sucesso!' });
        });
    } catch (erro) {
        res.status(500).json({ erro: 'Erro interno no servidor' });
    }
});

// Rota de LOGIN
app.post('/api/login', (req, res) => {
    const { email, senha } = req.body;

    const sql = 'SELECT * FROM usuarios WHERE email = ?';
    pool.query(sql, [email], async (err, results) => {
        if (err) return res.status(500).json({ erro: 'Erro no banco de dados' });

        // Verifica se o email existe
        if (results.length === 0) {
            return res.status(401).json({ erro: 'Usuário não encontrado' });
        }

        const usuario = results[0];

        // Compara a senha digitada com o hash salvo no banco
        const senhaValida = await bcrypt.compare(senha, usuario.senha);

        if (!senhaValida) {
            return res.status(401).json({ erro: 'Senha incorreta' });
        }

        // Devolve os dados do usuário (NUNCA devolva a senha)
        res.json({
            id: usuario.id,
            nome: usuario.nome,
            perfil_id: usuario.perfil_id
        });
    });
});

// 1. Função de Login simulada (você conectará isso ao botão de 'Entrar' no seu HTML)
async function fazerLogin(emailDigitado, senhaDigitada) {
    const resposta = await fetch('http://localhost:3000/api/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: emailDigitado, senha: senhaDigitada })
    });

    if (resposta.ok) {
        const dadosUsuario = await resposta.json();
        // Salva o usuário no navegador para ele não deslogar ao dar F5
        localStorage.setItem('usuarioLogado', JSON.stringify(dadosUsuario));
        carregarPainel(); // Monta a tela
    } else {
        alert("Email ou senha incorretos.");
    }
}
